        function signContract() {
            const anthonyInput = document.getElementById('sig-anthony').value.trim().toLowerCase();
            const florInput = document.getElementById('sig-flor').value.trim().toLowerCase();
            const errorMsg = document.getElementById('error-msg');
            const signingArea = document.getElementById('signing-area');
            const signedSeal = document.getElementById('signed-seal');
            const signBtn = document.getElementById('sign-btn');

            // Validación: Busca "anthony" Y "fonseca" en el primer campo, "flor" Y "castillo" en el segundo
            const validAnthony = anthonyInput.includes('anthony') && anthonyInput.includes('fonseca');
            const validFlor = florInput.includes('flor') && florInput.includes('castillo');

            if (validAnthony && validFlor) {
                // Success
                errorMsg.style.opacity = '0';
                
                // Disable inputs
                document.getElementById('sig-anthony').disabled = true;
                document.getElementById('sig-flor').disabled = true;
                document.getElementById('sig-anthony').classList.add('text-pink-700');
                document.getElementById('sig-flor').classList.add('text-pink-700');
                
                // Hide button
                signBtn.style.display = 'none';

                // Set date
                const today = new Date();
                const options = { year: 'numeric', month: 'long', day: 'numeric' };
                document.getElementById('sign-date').innerText = today.toLocaleDateString('es-ES', options);

                // Reveal Seal
                signedSeal.classList.remove('hidden');
                signedSeal.classList.add('flex');

                // Confetti Explosion
                fireConfetti();
                
                // Scroll to seal
                setTimeout(() => {
                    signedSeal.scrollIntoView({ behavior: 'smooth', block: 'center' });
                }, 100);

            } else {
                // Error
                errorMsg.style.opacity = '1';
                errorMsg.innerText = 'Para sellar este pacto, deben escribir exactamente: "Anthony Fonseca" y "Flor Castillo"';
                
                // Shake effect
                signingArea.classList.add('animate-pulse');
                setTimeout(() => signingArea.classList.remove('animate-pulse'), 500);
            }
        }

        function fireConfetti() {
            var duration = 3000;
            var end = Date.now() + duration;

            (function frame() {
                confetti({
                    particleCount: 5,
                    angle: 60,
                    spread: 55,
                    origin: { x: 0 },
                    colors: ['#db2777', '#fbcfe8', '#be185d'] 
                });
                confetti({
                    particleCount: 5,
                    angle: 120,
                    spread: 55,
                    origin: { x: 1 },
                    colors: ['#db2777', '#fbcfe8', '#be185d']
                });

                if (Date.now() < end) {
                    requestAnimationFrame(frame);
                }
            }());
        }
